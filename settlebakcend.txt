// --- HELPER FUNCTIONS ---

async function saveSeed(gameId, seedHex) {
    const { error } = await supabase
        .from('game_seeds')
        .insert([{ game_id: gameId.toString(), seed_hex: seedHex }]);
    
    if (error) {
        throw new Error(`Supabase Save Error: ${error.message}`);
    }
}

function toBase58String(value) {
    if (!value) return null;
    if (typeof value === 'string') return value;
    if (typeof value.toBase58 === 'function') return value.toBase58();
    if (value.pubkey && typeof value.pubkey.toBase58 === 'function') return value.pubkey.toBase58();
    return String(value);
}

async function saveSettlementResult(roundId, winner, prize) {
    const { error } = await supabase
        .from(SETTLEMENTS_TABLE)
        .insert([{ 
            roundid: roundId.toString(),
            winner,
            prize: prize.toString()
        }]);

    if (error) {
        throw new Error(`Supabase Settlement Save Error: ${error.message}`);
    }
}

async function deriveWinnerAndPrize(signature, playerOne, playerTwo) {
    const tx = await connection.getParsedTransaction(signature, {
        maxSupportedTransactionVersion: 0,
        commitment: 'confirmed'
    });

    if (!tx || !tx.meta) {
        return null;
    }

    const accountKeys = tx.transaction.message.accountKeys.map(toBase58String);
    const preBalances = tx.meta.preBalances || [];
    const postBalances = tx.meta.postBalances || [];
    const deltas = new Map();

    accountKeys.forEach((key, index) => {
        if (!key) return;
        const delta = (postBalances[index] || 0) - (preBalances[index] || 0);
        deltas.set(key, delta);
    });

    const p1 = playerOne.toBase58();
    const p2 = playerTwo.toBase58();
    const p1Delta = deltas.get(p1) || 0;
    const p2Delta = deltas.get(p2) || 0;

    if (p1Delta <= 0 && p2Delta <= 0) {
        return null;
    }

    if (p1Delta >= p2Delta) {
        return { winner: p1, prize: Math.max(p1Delta, 0) };
    }

    return { winner: p2, prize: Math.max(p2Delta, 0) };
}

async function getSeed(gameId) {
    const { data, error } = await supabase
        .from('game_seeds')
        .select('seed_hex')
        .eq('game_id', gameId.toString())
        .single();
        
    if (error || !data) return null;
    console.log(`🔑 Retrieved seed for Game ${gameId}`);
    return data.seed_hex;
}

app.post('/generate-game', async (req, res) => {
    try {
        const { gameId } = req.body;
        if (!gameId) return res.status(400).json({ error: "Missing gameId" });

        // 1. Generate Secret 32-byte seed
        const serverSeed = crypto.randomBytes(32);
        
        // 2. Create SHA-256 hash (matches Rust 'hash' function)
        const serverHash = crypto.createHash('sha256').update(serverSeed).digest();

        // 3. Store seed securely
        await saveSeed(gameId.toString(), serverSeed.toString('hex'));

        console.log(`🆕 Game Created: ${gameId} | Hash: ${serverHash.toString('hex').slice(0, 10)}...`);

        res.json({ serverHash: Array.from(serverHash) });
    } catch (e) {
        res.status(500).json({ error: e.message });
    }
});

async function settleGame(gamePda, playerOne, playerTwo, gameId) {
    try {
        // 1. Await the seed from Supabase (Crucial fix for your error)
        const hexSeed = await getSeed(gameId.toString());
        
        if (!hexSeed) {
          //  return console.log(`⚠️ No seed found for game ${gameId}, skipping.`);
        }

        const serverSeed = Buffer.from(hexSeed, 'hex');

        // 2. Build Instruction Data: [discriminator 2] [32b seed]
        const data = Buffer.alloc(33);
        data.writeUInt8(2, 0); 
        serverSeed.copy(data, 1);

        const instruction = new TransactionInstruction({
            keys: [
                // 1. The Authorized Settler (House Wallet)
                { pubkey: houseKeypair.publicKey, isSigner: true, isWritable: false },
                // 2. The Game State Account
                { pubkey: gamePda, isSigner: false, isWritable: true },
                // 3. Player One
                { pubkey: playerOne, isSigner: false, isWritable: true },
                // 4. Player Two
                { pubkey: playerTwo, isSigner: false, isWritable: true },
                // 5. Commission Vault
                { pubkey: COMMISSION_ACC, isSigner: false, isWritable: true },
            ],
            programId: PROGRAM_ID,
            data,
        });

        // 3. Send and Confirm Transaction
        const tx = new Transaction().add(instruction);
        
        // Using sendAndConfirmTransaction is better for bots to ensure the next 
        // scan doesn't try to settle the same game twice while one is pending.
        const signature = await connection.sendTransaction(tx, [houseKeypair], {
            skipPreflight: false,
            preflightCommitment: 'confirmed'
        });

        await connection.confirmTransaction(signature, 'confirmed');

        const settlement = await deriveWinnerAndPrize(signature, playerOne, playerTwo);
        if (!settlement) {
            throw new Error(`Unable to derive winner for game ${gameId}`);
        }
        
        console.log(`🎯 SETTLED Game ${gameId}. Sig: ${signature.slice(0, 10)}...`);
        console.log(`🏆 Winner: ${settlement.winner} | Prize: ${settlement.prize}`);

        await saveSettlementResult(gameId, settlement.winner, settlement.prize);
        console.log(`📝 Saved settlement for round ${gameId} to ${SETTLEMENTS_TABLE}`);

        // Keep the seed record so frontend history can still reconstruct
        // serverSeed and serverHash for settled games.

    } catch (err) {
        // Catch specific Solana errors (like 'already processed' if the bot double-fires)
        if (err.message.includes("already be processed")) {
             console.log(`ℹ️ Game ${gameId} already settled.`);
        } else {
           //  console.error(`❌ Settle Error (${gameId}):`, err.message);
        }
    }
}

async function scanForJoins() {
    try {
        // 1. Fetch all accounts owned by your program
        const accounts = await connection.getProgramAccounts(PROGRAM_ID);
        
        console.log(`🔍 Scanning ${accounts.length} accounts for active matches...`);

        for (const account of accounts) {
            const data = account.account.data;
            
            // Basic length check to avoid system/rent accounts
            if (data.length < 180) continue;

            // 2. Read the Status Byte (at index 81)
            const status = data.readUInt8(81); 

            // status === 2 means "Joined" (Ready for settlement)
            if (status === 2) { 
                // 3. Extract necessary PublicKeys and Game ID from the buffer
                const p1 = new PublicKey(data.slice(0, 32));
                const p2 = new PublicKey(data.slice(32, 64));
                const gId = data.readBigUInt64LE(73);

               // console.log(`🎰 Match Found! GameID: ${gId}. Starting settlement...`);

                // 4. CRITICAL: Use 'await' here.
                // This ensures the House wallet finishes one transaction 
                // before starting the next to prevent Nonce/Sequence errors.
                await settleGame(account.pubkey, p1, p2, gId);
            }
        }
    } catch (e) {
        console.error("❌ Scanner Error:", e.message);
    }
}
